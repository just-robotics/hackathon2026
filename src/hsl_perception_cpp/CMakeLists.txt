cmake_minimum_required(VERSION 3.8)
project(hsl_perception_cpp)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
foreach(dep rclcpp sensor_msgs nav_msgs std_msgs tf2_ros)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(opponent_detector src/node.cpp)
target_include_directories(opponent_detector PRIVATE include ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(opponent_detector rclcpp sensor_msgs nav_msgs std_msgs tf2_ros)
add_executable(detector_replay src/replay.cpp)
target_include_directories(detector_replay PRIVATE include ${EIGEN3_INCLUDE_DIRS})
install(TARGETS opponent_detector detector_replay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(detector_unit_tests test/detector_unit.cpp)
  target_include_directories(detector_unit_tests PRIVATE include ${EIGEN3_INCLUDE_DIRS})
  add_test(NAME detector_unit COMMAND detector_unit_tests)
endif()
ament_package()
